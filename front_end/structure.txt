│   │   └── bootstrap.min.css

│   │   └── bootstrap.bundle.min.js

│   ├── assets/                          ← للملفات الثابتة اللي ما بتتغيرش واللي بيستخدمها الكود جوه المشروع زي الصور.
│   │   └── images/                      ← بيجمع صور الموقع مقسّمة جوه فولدرات على اسم كل صفحة.

│   ├── pages/                           ← بيحتوي على كل صفحات المشروع الأساسية (`Home` و `Profile` و `NotFound`).
│   │   ├── Home/                        ← (مثال لصفحة واحدة) الفولدر الجامع لكل الملفات الخاصة بصفحة `Home` فقط.
│   │   │   ├── components/              ← بيجمع المكونات الفرعية الخاصة بصفحة `Home` لوحدها ومش بتستخدم برّاها.
│   │   │   │   └── HeroSection/         ← مكون فرعي جوه صفحة `Home` مع ملف الـ CSS الخاص بيه.

│   │   │   ├── Home.css

│   │       │       ├── UserInfoCard.css
│   │       │       └── UserInfoCard.jsx
│   │       ├── Profile.css
│   │       └── Profile.jsx
│   ├── serveses/                        ← مخصص لدوال التعامل مع الـ API وطلبات السيرفر بعيد عن ملفات العرض (JSX).
│   ├── shared/                          ← فيه المكونات المشتركة اللي بتتكرر في أكتر من صفحة زي الهيدر والفوتر.
│   │   ├── Footer/
│   │   │   ├── Footer.css
│   │   │   └── Footer.jsx
│   │   └── Header/
│   │       ├── Header.css
│   │       └── Header.jsx
│   ├── main.css                         ← ملف التنسيقات العامة على مستوى التطبيق كله.
│   └── main.jsx                         ← نقطة البداية اللي بتجمع كل حاجة في React وتحقنها في عنصر `root` جوه `index.html`.
├── index.html                           ← صفحة الـ HTML الوحيدة اللي بتستقبل التطبيق النهائي جوه `<div id="root"></div>`.
